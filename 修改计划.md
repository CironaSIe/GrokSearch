### 1.2 premise_valid 兼容性修复

**文件**: `src/grok_search/server.py:678`
**来源**: MoonWeSif
**改动**: 签名 `Optional[bool] = None` → `bool = True`，逻辑从 `is not None` → `not premise_valid`

| | 旧 | 新 |
|---|---|---|
| 类型 | `Optional[bool]` | `bool` |
| 默认值 | `None` | `True` |
| 条件 | `if premise_valid is not None` | `if not premise_valid` |

**行为变更**:
- `premise_valid=True`（显式或默认）→ 不传 `premise_valid` 字段（JSON Schema 中该字段变为可选）

- 旧版 `None` 现在等同 `True`——这是期望行为，因为 LLM 传 `premise_valid` 只有两种意图：显式标记某假设有问题时传 `false`，其余情况不传

### 1.5 流式解析错误追踪

**文件**: `src/grok_search/providers/grok.py:174-213`
**来源**: Techd81
**改动**:

`_parse_streaming_response` 增加错误计数 + 空结果显式抛异常：

```python
async def _parse_streaming_response(self, response, ctx=None) -> str:
    content = ""
    full_body_buffer = []
    parse_errors = 0

    async for line in response.aiter_lines():
        line = line.strip()
        if not line:
            continue
        full_body_buffer.append(line)
        if line.startswith("data:"):
            if line in ("data: [DONE]", "data:[DONE]"):
                continue
            try:
                json_str = line[5:].lstrip()
                data = json.loads(json_str)
                choices = data.get("choices", [])
                if choices and len(choices) > 0:
                    delta = choices[0].get("delta", {})
                    if "content" in delta:
                        content += delta["content"]
            except (json.JSONDecodeError, IndexError):
                parse_errors += 1
                continue

    if not content and full_body_buffer:
        try:
            full_text = "".join(full_body_buffer)
            data = json.loads(full_text)
            if "choices" in data and len(data["choices"]) > 0:
                message = data["choices"][0].get("message", {})
                content = message.get("content", "")
        except json.JSONDecodeError:
            parse_errors += 1

    if not content:
        if full_body_buffer:
            raise ValueError(
                "Grok stream parse error: no content found"
                f" (parse_errors={parse_errors})"
            )
        raise ValueError("Grok stream parse error: empty streaming response")

    return content
```

**设计依据**: 当前代码在 `_parse_streaming_response` 返回空字符串时，`_execute_stream_with_retry` 静默返回 `""`，导致 `web_search` 返回 `answer=""` 而用户看到"空搜索"。这步让空流**尽早失败**。

**配套修复**: `_safe_grok` 增加 `except ValueError` 分支将错误信息透传给用户（而非静默吞掉）。阶段 2 将在此基础上增加完整错误格式化（`_format_grok_error`）+ 非流式降级（`_execute_non_stream_with_retry`）。

**验证**: `python -c "from grok_search.server import web_search; import inspect; assert '[Grok' in inspect.getsource(web_search)"` 确认 `_safe_grok` 有 ValueError 处理。

### 2.1 API Key 脱敏

```python
def redact_sensitive_text(text: str, api_key: str = "") -> str:
    redacted = str(text or "")
    if api_key:
        redacted = redacted.replace(api_key, "***")
    redacted = re.sub(
        r"(?i)(authorization\s*[:=]\s*bearer\s+)[^\s,;}]+",
        r"\1***", redacted,
    )
    redacted = re.sub(
        r"(?i)((?:api[_-]?key|token|secret)\s*[:=]\s*)[^\s,;}]+",
        r"\1***", redacted,
    )
    return redacted
```

**设计依据**: `get_config_info` 已有 `_mask_api_key` 做配置脱敏，但错误日志和 `_safe_grok` 返回的异常文本可能暴露完整 Key。此函数用于所有错误路径的文本清理。

```python
def _format_grok_error(exc: Exception, api_key: str = "") -> str:
    if isinstance(exc, httpx.HTTPStatusError):
        body = redact_sensitive_text(exc.response.text or "", api_key).strip()[:500]
        detail = f": {body}" if body else ""
        return f"Grok 调用失败: HTTP {exc.response.status_code}{detail}"

    if isinstance(exc, httpx.RequestError):
        msg = redact_sensitive_text(str(exc), api_key).strip()
        cls_name = exc.__class__.__name__
        return f"Grok 调用失败: 网络错误 ({cls_name})" + (f": {msg}" if msg else "")

    if isinstance(exc, ValueError):
        return f"Grok 调用失败: {exc}"

    msg = redact_sensitive_text(str(exc), api_key).strip()
    return f"Grok 调用失败: {exc.__class__.__name__}" + (f": {msg}" if msg else "")
```

```python
async def _execute_stream_with_retry(self, headers, payload, ctx=None) -> str:
    timeout = httpx.Timeout(connect=6.0, read=120.0, write=10.0, pool=None)
    async with httpx.AsyncClient(timeout=timeout, follow_redirects=True) as client:
```

```python
async def _execute_non_stream_with_retry(self, headers, payload, ctx=None) -> str:
    body = dict(payload)
    body["stream"] = False
    timeout = httpx.Timeout(connect=6.0, read=120.0, write=10.0, pool=None)
    async with httpx.AsyncClient(timeout=timeout, follow_redirects=True) as client:
        async for attempt in AsyncRetrying(
            stop=stop_after_attempt(config.retry_max_attempts + 1),
            wait=_WaitWithRetryAfter(config.retry_multiplier, config.retry_max_wait),
            retry=retry_if_exception(_is_retryable_exception),
            reraise=True,
        ):
            with attempt:
                response = await client.post(
                    f"{self.api_url}/chat/completions",
                    headers=headers, json=body,
                )
                response.raise_for_status()
                data = response.json()
                return self._extract_content_from_completion(data)

@staticmethod
def _extract_content_from_completion(data: dict) -> str:
    choices = data.get("choices", [])
    if not choices:
        return ""
    message = choices[0].get("message", {})
    content = message.get("content", "")
    if isinstance(content, str):
        return content
    if isinstance(content, list):
        parts = [item["text"] for item in content
                 if isinstance(item, dict) and item.get("type") == "text"]
        return "".join(parts)
    return ""
```

**设计依据**:
- 降级在 retry 全部耗尽后触发，不影响正常流式路径
- xAI API 兼容 OpenAI Chat Completions 格式，非流式 response 的 `choices[0].message.content` 可稳定解析
- 降级同样带 retry（部分网关只对 streaming 返回空，non-streaming 正常）

### 2.4 Tavily API Key 轮换（真 round-robin）

**文件**: `src/grok_search/config.py`, `src/grok_search/server.py`
**来源**: handsomelong922（概念），重新设计实现
**设计差异说明**: 原 fork 实现是顺序 fallback（一次请求试完所有 key），我改为真 round-robin（每次请求取一个 key，失败不等重试，下次请求自动取下一个）。理由：轮换是为了分摊配额限制，fallback 会更快耗尽所有 key 的配额。

**config.py**：

```python
# __new__ 新增
cls._instance._tavily_key_index = 0
cls._instance._tavily_key_lock = asyncio.Lock()

# 新增属性
@property
def tavily_api_key_list(self) -> list[str]:
    raw = os.getenv("TAVILY_API_KEY")
    if not raw:
        return []
    return [k.strip() for k in raw.split(",") if k.strip()]

async def get_next_tavily_api_key(self) -> str | None:
    keys = self.tavily_api_key_list
    if not keys:
        return None
    async with self._tavily_key_lock:
        key = keys[self._tavily_key_index % len(keys)]
        self._tavily_key_index += 1
        return key
```

**server.py** 三个 `_call_tavily_*` 函数改为：


```python
async def search(self, query, platform="", from_date="", to_date="",
                 allowed_domains="", max_search_results=0, min_results=3,
                 max_results=10, ctx=None) -> List[SearchResult]:
    # 构建 search_controls prompt 片段
    controls = []
    if allowed_domains:
        controls.append(f"Only search these domains: {allowed_domains}.")
    if max_search_results > 0:
        controls.append(f"Use at most {max_search_results} results.")

    user_content = time_context + query + platform_prompt
    if controls:
        user_content += "\n\n[Search Constraints]\n" + "\n".join(f"- {c}" for c in controls)

    payload = {..., "messages": [{"role": "system", "content": search_prompt},
                                  {"role": "user", "content": user_content}], ...}

    # 如有 from/to date，注入 system prompt（Grok API search_parameters 待验证后加）
    # 当前方案：作为额外约束写入 system prompt
    date_constraint = ""
    if from_date or to_date:
        date_range = f"from {from_date}" if from_date else ""
        if to_date:
            date_range += f" to {to_date}" if date_range else f"before {to_date}"
        date_constraint = f"\nFocus on results {date_range.strip()}."
    if date_constraint:
        payload["messages"][0]["content"] += date_constraint
```

```python
        super().__init__(api_url, api_key)
        self.model = model
        self._reasoning_prompt = self._build_reasoning_prompt(reasoning_effort)

    @staticmethod
    def _build_reasoning_prompt(effort: str) -> str:
        effort = effort.strip().lower()
        if effort == "low":
            return "Be concise; minimal analysis."
        elif effort == "high":
            return "Think step by step; show your reasoning."
        elif effort == "xhigh":
            return "Very detailed reasoning; explore multiple angles."
        else:
            return ""  # medium or invalid = default
```

`self._reasoning_prompt` 在 `search()` 中注入到 system message 末尾。

### 3.4 Multi-Agent 模型识别 + Responses API（保留到 v2.0+ 评估）

当前计划**不实现** multi-agent / Responses API 双模路由。理由：

1. **xAI Responses API 端点未知**——当前代码仅使用 `/chat/completions`，Responses API 端点（可能是 `/v1/responses`）需要 xAI 文档确认
2. **流事件格式不同**——Responses API 的 SSE 事件流格式与 OpenAI Chat Completions 格式不同，解析器需要全新开发
3. **官方代理无必要降级**——`api.x.ai` 官方 URL 总是支持 Chat Completions 和 Responses API 双模
4. **优先级低**——当前 `grok-4.20-fast` + `grok-4.20-vision` 模型在 Chat Completions 下运行正常

**替代方案**：如果用户需要 `reasoning_effort`，通过 3.3 的 prompt 注入方式实现近似效果，不引入新的 API 端点。

```python
def _format_sources_markdown(sources: list[dict]) -> str:
    lines = []
    for item in sources:
        url = item.get("url", "")
        title = item.get("title") or url
        # 转义 markdown 特殊字符
```

```python
"""Exa search API provider."""
import httpx
import json
from typing import Optional
from .base import BaseSearchProvider, SearchResult


class ExaSearchProvider(BaseSearchProvider):
    def __init__(self, api_key: str, base_url: str = "https://api.exa.ai"):
        super().__init__(base_url, api_key)
        self.base_url = base_url.rstrip("/")

    def get_provider_name(self) -> str:
        return "Exa"

    async def search(self, query: str, max_results: int = 5,
                     include_domains: list[str] | None = None,
                     exclude_domains: list[str] | None = None,
                     category: str | None = None) -> list[SearchResult]:
        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "Content-Type": "application/json",
        }
        body: dict = {"query": query, "num_results": max_results}
        if include_domains:
            body["include_domains"] = include_domains
        if exclude_domains:
            body["exclude_domains"] = exclude_domains
        if category:
            body["category"] = category

        async with httpx.AsyncClient(timeout=30.0) as client:
            response = await client.post(
                f"{self.base_url}/search",
                headers=headers, json=body,
            )
            response.raise_for_status()
            data = response.json()

        results = []
        for item in (data.get("results", []) or []):
            results.append(SearchResult(
                title=item.get("title", ""),
                url=item.get("url", ""),
                snippet=item.get("text", "") or item.get("snippet", ""),
                source="exa",
                published_date=item.get("publishedDate", ""),
            ))
        return results

    async def find_similar(self, url: str, num_results: int = 5) -> list[SearchResult]:
        """Exa findSimilar API — 找与指定 URL 语义相似的页面"""
        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "Content-Type": "application/json",
        }
        body = {"url": url, "num_results": num_results}
        async with httpx.AsyncClient(timeout=30.0) as client:
            response = await client.post(
                f"{self.base_url}/findSimilar",
                headers=headers, json=body,
            )
            response.raise_for_status()
            data = response.json()
        results = []
        for item in (data.get("results", []) or []):
            results.append(SearchResult(
                title=item.get("title", ""),
                url=item.get("url", ""),
                snippet=item.get("text", "") or item.get("snippet", ""),
                source="exa",
                published_date=item.get("publishedDate", ""),
            ))
        return results
```
